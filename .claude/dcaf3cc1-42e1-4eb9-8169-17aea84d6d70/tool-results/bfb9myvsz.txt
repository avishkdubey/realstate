     1	"use client";
     2	
     3	import { useEffect, useRef, useState } from "react";
     4	
     5	import { useReducedMotionPreference } from "@/components/providers/reduced-motion-provider";
     6	
     7	type Frame = ImageBitmap | HTMLImageElement;
     8	
     9	/**
    10	 * Scroll-driven image sequence — the technique Apple uses for product pages.
    11	 *
    12	 * A video is decoded to numbered stills ahead of time (see
    13	 * `scripts/extract-frames.md`), and scroll position selects which still is
    14	 * painted to a canvas. Scrubbing a sequence this way is far more responsive
    15	 * than seeking a `<video>`, which forces a decode per seek and stutters badly
    16	 * on mobile.
    17	 *
    18	 * Design notes:
    19	 *  - frames live in state, not a ref, and the render loop is a single effect
    20	 *    that depends on them. An earlier version split loading and painting
    21	 *    across two effects communicating through refs, and the loop could end up
    22	 *    reading a cache the other effect had already replaced — which froze the
    23	 *    sequence on one frame with no error anywhere.
    24	 *  - scroll position is sampled inside rAF rather than in a scroll listener,
    25	 *    so it stays in step with Lenis (which drives scrolling from its own loop)
    26	 *    and a fast flick never queues hundreds of draws.
    27	 *  - frame 1 renders immediately as a plain `<img>`, so the hero is never
    28	 *    empty while the rest decode.
    29	 *  - reduced motion, Data Saver and 2G keep that still and skip the sequence.
    30	 */
    31	export function ScrollFrameSequence({
    32	  frameCount,
    33	  framePath,
    34	  poster,
    35	  className,
    36	  children,
    37	}: {
    38	  frameCount: number;
    39	  /** Given a 1-based index, returns that frame's URL. */
    40	  framePath: (index: number) => string;
    41	  /** Rendered immediately and whenever the sequence is skipped. */
    42	  poster: string;
    43	  className?: string;
    44	  children?: React.ReactNode;
    45	}) {
    46	  const section = useRef<HTMLDivElement>(null);
    47	  const canvas = useRef<HTMLCanvasElement>(null);
    48	
    49	  const reducedMotion = useReducedMotionPreference();
    50	  const [frames, setFrames] = useState<Frame[]>([]);
    51	  /**
    52	   * The canvas only fades in once something has actually been drawn on it.
    53	   * Revealing it as soon as the frames decoded showed a black rectangle over
    54	   * the poster on any tab where rAF had not run yet — a backgrounded tab, for
    55	   * instance, where rAF is paused entirely.
    56	   */
    57	  const [hasPainted, setHasPainted] = useState(false);
    58	
    59	  /* ---------------------------------------------------------------- load */
    60	  useEffect(() => {
    61	    if (reducedMotion) return;
    62	
    63	    // A frame sequence is decoration; decoration should never cost someone
    64	    // their data allowance.
    65	    const connection = (
    66	      navigator as { connection?: { saveData?: boolean; effectiveType?: string } }
    67	    ).connection;
    68	    if (connection?.saveData) return;
    69	    if (connection?.effectiveType && /(^|-)2g/.test(connection.effectiveType)) return;
    70	
    71	    let cancelled = false;
    72	
    73	    const load = async () => {
    74	      const settled = await Promise.all(
    75	        Array.from({ length: frameCount }, (_, i) =>
    76	          loadFrame(framePath(i + 1)).catch(() => null),
    77	        ),
    78	      );
    79	      if (cancelled) return;
    80	
    81	      const usable = settled.filter((frame): frame is Frame => frame !== null);
    82	      if (usable.length > 0) setFrames(usable);
    83	    };
    84	
    85	    // Decode after first paint so the sequence never competes with the LCP.
    86	    const schedule =
    87	      window.requestIdleCallback ?? ((cb: IdleRequestCallback) => setTimeout(cb, 600));
    88	    const handle = schedule(() => void load());
    89	
    90	    return () => {
    91	      cancelled = true;
    92	      if (window.cancelIdleCallback && typeof handle === "number") {
    93	        window.cancelIdleCallback(handle);
    94	      }
    95	    };
    96	  }, [frameCount, framePath, reducedMotion]);
    97	
    98	  /* --------------------------------------------------------------- paint */
    99	  useEffect(() => {
   100	    if (reducedMotion || frames.length === 0) return;
   101	
   102	    const node = section.current;
   103	    const surface = canvas.current;
   104	    if (!node || !surface) return;
   105	
   106	    const context = surface.getContext("2d", { alpha: false });
   107	    if (!context) return;
   108	
   109	    let painted = -1;
   110	    let raf = 0;
   111	
   112	    const draw = (frame: Frame) => {
   113	      // Cover-fit, so the sequence behaves like a background image.
   114	      const { width: cw, height: ch } = surface;
   115	      const scale = Math.max(cw / frame.width, ch / frame.height);
   116	      const dw = frame.width * scale;
   117	      const dh = frame.height * scale;
   118	      context.drawImage(frame, (cw - dw) / 2, (ch - dh) / 2, dw, dh);
   119	    };
   120	
   121	    /** Reads scroll position and paints the matching frame. Returns if unchanged. */
   122	    const render = () => {
   123	      const rect = node.getBoundingClientRect();
   124	      const travel = rect.height - window.innerHeight;
   125	      if (travel <= 0) return;
   126	
   127	      const progress = Math.min(1, Math.max(0, -rect.top / travel));
   128	      const index = Math.min(
   129	        frames.length - 1,
   130	        Math.round(progress * (frames.length - 1)),
   131	      );
   132	      if (index === painted) return;
   133	
   134	      draw(frames[index]);
   135	      painted = index;
   136	      // Exposed so the sequence can be asserted on from outside.
   137	      surface.dataset.frame = String(index);
   138	      setHasPainted(true);
   139	    };
   140	
   141	    const tick = () => {
   142	      raf = requestAnimationFrame(tick);
   143	      render();
   144	    };
   145	
   146	    const resize = () => {
   147	      const dpr = Math.min(window.devicePixelRatio, 1.5);
   148	      surface.width = Math.round(surface.clientWidth * dpr);
   149	      surface.height = Math.round(surface.clientHeight * dpr);
   150	      painted = -1; // resizing a canvas clears it
   151	      render();
   152	    };
   153	
   154	    // Paint once synchronously, so the canvas is never revealed empty even if
   155	    // the first animation frame is delayed or the tab is not rendering yet.
   156	    resize();
   157	    raf = requestAnimationFrame(tick);
   158	    window.addEventListener("resize", resize);
   159	
   160	    return () => {
   161	      cancelAnimationFrame(raf);
   162	      window.removeEventListener("resize", resize);
   163	    };
   164	  }, [frames, reducedMotion]);
   165	
   166	  const active = hasPainted && !reducedMotion;
   167	
   168	  return (
   169	    <div ref={section} className={className}>
   170	      <div className="sticky top-0 h-svh w-full overflow-hidden">
   171	        {/* Always present, so there is never an empty frame. */}
   172	        {/* eslint-disable-next-line @next/next/no-img-element */}
   173	        <img
   174	          src={poster}
   175	          alt=""
   176	          aria-hidden
   177	          fetchPriority="high"
   178	          className="absolute inset-0 h-full w-full object-cover"
   179	        />
   180	        <canvas
   181	          ref={canvas}
   182	          aria-hidden
   183	          className={`absolute inset-0 h-full w-full transition-opacity duration-700 ${
   184	            active ? "opacity-100" : "opacity-0"
   185	          }`}
   186	        />
   187	        {children}
   188	      </div>
   189	    </div>
   190	  );
   191	}

   193	/** Prefers ImageBitmap — decoding off the main thread keeps scrolling smooth. */
   194	async function loadFrame(url: string): Promise<Frame> {
   195	  if (typeof createImageBitmap === "function") {
   196	    const response = await fetch(url);
   197	    if (!response.ok) throw new Error(`frame ${url} failed`);
   198	    return createImageBitmap(await response.blob());
   199	  }
   200	
   201	  return new Promise((resolve, reject) => {
   202	    const image = new Image();
   203	    image.onload = () => resolve(image);
   204	    image.onerror = reject;
   205	    image.src = url;
   206	  });
   207	}

   131	function TowerMesh({
   132	  tower,
   133	  index,
   134	  count,
   135	  selected,
   136	  onSelect,
   137	}: {
   138	  tower: Tower;
   139	  index: number;
   140	  count: number;
   141	  selected: boolean;
   142	  onSelect: () => void;
   143	}) {
   144	  const mesh = useRef<THREE.Mesh>(null);
   145	  const [hovered, setHovered] = useState(false);
   146	
   147	  // Height tracks the real floor count so the model is at least directionally
   148	  // honest about which tower is taller.
   149	  const height = Math.max(2, tower.floors * 0.28);
   150	  const spread = 3.2;
   151	  const x = (index - (count - 1) / 2) * spread;
   152	
   153	  // Each tower owns its material so hover state cannot leak between them.
   154	  const material = useMemo(
   155	    () =>
   156	      new THREE.MeshStandardMaterial({
   157	        color: "#2a2622",
   158	        roughness: 0.35,
   159	        metalness: 0.55,
   160	      }),
   161	    [],
   162	  );
   163	
   164	  // Target colours are built once. Allocating a THREE.Color inside the frame
   165	  // loop is the classic way to turn a smooth scene into a stuttering one —
   166	  // it hands the garbage collector work sixty times a second (CLAUDE.md §7).
   167	  const palette = useMemo(
   168	    () => ({
   169	      idle: new THREE.Color("#2a2622"),
   170	      hover: new THREE.Color("#8c6a3f"),
   171	      selected: new THREE.Color("#b99c6b"),
   172	    }),
   173	    [],
   174	  );
   175	
   176	  useFrame(() => {
   177	    if (!mesh.current) return;
   178	    const target = selected
   179	      ? palette.selected
   180	      : hovered
   181	        ? palette.hover
   182	        : palette.idle;
   183	    (mesh.current.material as THREE.MeshStandardMaterial).color.lerp(target, 0.12);
   184	  });
   185	
   186	  return (
   187	    <mesh
   188	      ref={mesh}
   189	      position={[x, height / 2, 0]}
   190	      material={material}
   191	      onClick={(event) => {
   192	        event.stopPropagation();
   193	        onSelect();
   194	      }}
   195	      onPointerOver={(event) => {
   196	        event.stopPropagation();
   197	        setHovered(true);
   198	        document.body.style.cursor = "pointer";
   199	      }}
   200	      onPointerOut={() => {
   201	        setHovered(false);
   202	        document.body.style.cursor = "";
   203	      }}
   204	    >
   205	      <boxGeometry args={[1.6, height, 1.6]} />
   206	    </mesh>
   207	  );
   208	}

    13	/**
    14	 * 360° panorama, rendered with Pannellum.
    15	 *
    16	 * Matters most to buyers who decide without ever standing in the building,
    17	 * which describes a large share of NRI purchases (CLAUDE.md §2, §5).
    18	 *
    19	 * Pannellum is a UMD bundle that attaches to `window` and needs a DOM, so it
    20	 * is imported inside an effect rather than at module scope — and only once the
    21	 * viewer is near the viewport, so neither the script nor the panorama image
    22	 * costs anything on pages the visitor never scrolls.
    23	 */
    24	export function PannellumTour({
    25	  src,
    26	  label,
    27	}: {
    28	  /** Equirectangular (2:1) panorama image. */
    29	  src: string;
    30	  label: string;
    31	}) {
    32	  const container = useRef<HTMLDivElement>(null);
    33	  const [near, setNear] = useState(false);
    34	  const [failed, setFailed] = useState(false);
    35	
    36	  useEffect(() => {
    37	    const node = container.current;
    38	    if (!node) return;
    39	
    40	    const observer = new IntersectionObserver(
    41	      ([entry]) => {
    42	        if (entry.isIntersecting) {
    43	          setNear(true);
    44	          observer.disconnect();
    45	        }
    46	      },
    47	      { rootMargin: "200px" },
    48	    );
    49	
    50	    observer.observe(node);
    51	    return () => observer.disconnect();
    52	  }, []);
    53	
    54	  useEffect(() => {
    55	    if (!near || !container.current) return;
    56	
    57	    const node = container.current;
    58	    let viewer: PannellumViewer | undefined;
    59	    let disposed = false;
    60	
    61	    (async () => {
    62	      try {
    63	        await import("pannellum/build/pannellum.css");
    64	        await import("pannellum/build/pannellum.js");
    65	        if (disposed) return;
    66	
    67	        const pannellum = (window as unknown as { pannellum?: PannellumApi })
    68	          .pannellum;
    69	        if (!pannellum) throw new Error("pannellum failed to attach to window");
    70	
    71	        viewer = pannellum.viewer(node, {
    72	          type: "equirectangular",
    73	          panorama: src,
    74	          autoLoad: true,
    75	          // A slow drift reads as live without hijacking the visitor's scroll.
    76	          autoRotate: -2,
    77	          autoRotateInactivityDelay: 3000,
    78	          showZoomCtrl: true,
    79	          showFullscreenCtrl: true,
    80	          keyboardZoom: true,
    81	          mouseZoom: "fullscreenonly",
    82	          backgroundColor: [0.08, 0.08, 0.08],
    83	        });
    84	      } catch (error) {
    85	        console.warn("[pannellum] failed to initialise", error);
    86	        setFailed(true);
    87	      }
    88	    })();
    89	
    90	    return () => {
    91	      disposed = true;
    92	      try {
    93	        viewer?.destroy();
    94	      } catch {
    95	        // Pannellum throws if it never finished initialising; nothing to clean.
    96	      }
    97	    };
    98	  }, [near, src]);
